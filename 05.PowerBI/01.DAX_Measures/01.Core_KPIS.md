
-- 1. Total Revenue
-------------------
Total Revenue = CALCULATE(SUM(order_items[price]) , orders[order_status] ="delivered")


-- 2. Total Orders
------------------
Total Orders = COUNTROWS(orders)


-- 3. Total Customres
---------------------
Total Customres = DISTINCTCOUNT(customers[customer_unique_id])


-- 4. Delivered Orders
----------------------
Delivered Orders = CALCULATE(COUNTROWS(orders) , orders[order_status]="delivered")


-- 5. Average Order Value 
-------------------------
Average Order Value = DIVIDE([Total Revenue] , [Delivered Orders])
