
-- 1. Total Revenue
-------------------
Total Revenue = CALCULATE(SUM(order_items[price]) , orders[order_status] ="delivered")


-- 2. Total Orders
------------------
Total Orders = COUNTROWS(orders)


-- 3. Total Customres
---------------------
Total Customres = DISTINCTCOUNT(customers[customer_unique_id])


-- 4. Delivered Orders
----------------------
Delivered Orders = CALCULATE(COUNTROWS(orders) , orders[order_status]="delivered")


-- 5. Cancelled Orders
----------------------
Cancelled Orders = CALCULATE(COUNTROWS(orders) , orders[order_status]="cancelled")


-- 6. Average Order Value 
-------------------------
Average Order Value = DIVIDE([Total Revenue] , [Delivered Orders])


-- 7. Successful Order Rate
---------------------------
Successful Order Rate  = DIVIDE([Delivered Orders],[Total Orders])


-- 8. Average Review Score
--------------------------
Average Review Score = AVERAGE(order_reviews[review_score])
